// Capture before async boot, then erase even malformed fragments. Never send
// the PIN in a GET/query, persist it, or treat opening a URL as confirmation.
function takeLineConfirmLink() {
  const hash = location.hash;
  if (!hash) return null;
  try { history.replaceState(null, "", location.pathname + location.search); }
  catch { return null; }
  if (!lineReturnQuery) return null;
  const parts = new URLSearchParams(hash.slice(1));
  if (parts.size !== 3 || ["line-confirm", "store", "expires"].some(k => parts.getAll(k).length !== 1)) return null;
  const pin = parts.get("line-confirm"), storeId = parts.get("store"), rawExpiry = parts.get("expires");
  const expiresAt = Number(rawExpiry);
  if (!/^\d{6}$/.test(pin ?? "") || !/^[A-Za-z0-9_-]{1,128}$/.test(storeId ?? "") ||
      !/^[1-9]\d*$/.test(rawExpiry ?? "") || !Number.isSafeInteger(expiresAt) || expiresAt <= Date.now()) return null;
  return {pin, storeId, expiresAt};
}

function expireLineConfirmLink() {

function armLineConfirmLinkExpiry() {
  clearTimeout(lineConfirmLinkTimer);

    rendered = "";
    if (state) render().catch(showError);
  }, Math.max(0, lineConfirmLink.expiresAt - Date.now()));
}

async function api(path, body) {
  if (body !== undefined) discardLineConfirmLink();
  const response = await fetch("/api/self/" + path, {
    method: body === undefined ? "GET" : "POST",
    headers:
      body === undefined
        ? {}
        : {
            "Content-Type": "application/json",
            "X-CSRF-Token": state?.csrf ?? "",
          },
    body: body === undefined ? undefined : JSON.stringify(body),
    credentials: "same-origin",
  });
  const data = await response.json();
  if (!response.ok || !data.ok) throw data;
  return data;
}
// One-button mode: remember which check was sent as a number-only text, so the
// guidance does not point at the card that this send invalidated. Only expiry
// timestamps are kept (never the number); a new card send replaces them.
// Stored forms: "<expiresAt>" (number-only send accepted), "<expiresAt>:unknown"
// (a number-only send whose result is unknown issued this check) and
// "from:<previous expiresAt or 0>" (a number-only send started; the server
// replaces the previous check before dispatch, so any newer check came from it).
const PIN_ONLY_KEY = "meo-line-pin-only";
let pinOnlyExpiresAt = null, pinOnlyUnknown = false, pinOnlyFrom = null;
try {
  const saved = String(sessionStorage.getItem(PIN_ONLY_KEY) ?? "");
  const from = /^from:(\d+)$/.exec(saved), at = /^(\d+)(:unknown)?$/.exec(saved);
  if (from) pinOnlyFrom = Number(from[1]);
  else if (at) { pinOnlyExpiresAt = Number(at[1]) || null; pinOnlyUnknown = Boolean(pinOnlyExpiresAt && at[2]); }
} catch {}
function storePinOnly() {
  try {
    if (pinOnlyFrom !== null) sessionStorage.setItem(PIN_ONLY_KEY, "from:" + pinOnlyFrom);
    else if (pinOnlyExpiresAt) sessionStorage.setItem(PIN_ONLY_KEY, pinOnlyExpiresAt + (pinOnlyUnknown ? ":unknown" : ""));
    else sessionStorage.removeItem(PIN_ONLY_KEY);
  } catch {}
}
function rememberPinOnly(expiresAt, unknown = false) {
  pinOnlyExpiresAt = expiresAt || null;
  pinOnlyUnknown = Boolean(pinOnlyExpiresAt && unknown);
  pinOnlyFrom = null;
  storePinOnly();
}
function startPinOnly(previousExpiresAt) {
  pinOnlyFrom = Number(previousExpiresAt) || 0;
  storePinOnly();
}
// A card send or a new registration code must not inherit an unresolved
// number-only attempt, or its valid card would be shown as that attempt's
// result. A settled marker stays: it only matches its own check, so a replaced
// check ignores it.
function forgetPinOnlyAttempt() {
  if (pinOnlyFrom === null) return;
  pinOnlyFrom = null;
  storePinOnly();
}
// After a timeout or a lost response the browser never learns the send's
// result. A check newer than the one seen before that send was issued by it.
function bindPinOnly(check) {
  if (pinOnlyFrom !== null && check?.pinAvailable && check.expiresAt && check.expiresAt !== pinOnlyFrom)
    rememberPinOnly(check.expiresAt, true);
}

async function loadChallenge() {
  if (!state.turnstileSiteKey || challengeToken) return;
  const target = document.querySelector("#challenge");
  if (target.dataset.loaded) return;
  target.dataset.loaded = "true";
  const render = () => {
    challengeWidget = window.turnstile.render(target, {
      sitekey: state.turnstileSiteKey,
      action: "self_start",
      callback: (t) => {
        challengeToken = t;
        challengeStatus("");
        document
          .querySelectorAll("[data-oauth]")
          .forEach((b) => (b.disabled = busy));
      },
      "expired-callback": () => { clearChallenge(); challengeStatus(CHALLENGE_RETRY); },
      "error-callback": () => { clearChallenge(); challengeStatus(CHALLENGE_RETRY); },
    });
  };
  if (window.turnstile) return render();
  const script = document.createElement("script");
  script.src =
    "https://challenges.cloudflare.com/turnstile/v0/api.js?render=explicit";
  script.onload = render;
  script.onerror = () => {
    // The alert carries the reload instruction; do not repeat it in the status line.
    challengeLoadFailed = true;
    challengeStatus("");
    showError({ code: "CHALLENGE_LOAD_FAILED" });
  };
  document.head.append(script);
}
function clearChallenge() {
  challengeToken = "";
  document.querySelectorAll("[data-oauth]").forEach((b) => (b.disabled = true));
}

async function google(intent) {
  if (!challengeToken) throw { code: "CHALLENGE_FAILED" };
  const token = challengeToken;
  clearChallenge();
  try {
    const r = await api("google/start", { intent, challenge: token });
    location.assign(r.authorizationUrl);
  } catch (error) {
    clearChallenge();
    if (window.turnstile && challengeWidget !== null)
      window.turnstile.reset(challengeWidget);
    if (!challengeToken) challengeStatus(CHALLENGE_WAIT);
    throw error;
  }
}
async function refresh(force = false) {
  // Local expiry must clear the proof even if this GET later fails/offline.
  expireLineConfirmLink();
  const previous = state;
  state = await api(returningFromLine ? "line/return-status" : "status");
  if (state.lineOneButton) bindPinOnly(state.lineCheck);
  if (lineConfirmLink && !liveLineConfirmLink()) discardLineConfirmLink();
  if (state.lineDetected || !["location_selected", "line_pending"].includes(state.state)) {
    lineCode = null;
    lineCodeExpiresAt = 0;
  }
  if (state.lineAutoPin && state.lineDetected && state.lineCheck?.state === "idle") {
    lineIdleSince ??= Date.now();
    state.lineCheck = {...state.lineCheck,
      state: Date.now()-lineIdleSince < 90000 ? "awaiting" : "held",
      code: Date.now()-lineIdleSince < 90000 ? null : "LINE_RETRY_REQUIRED"};
  } else lineIdleSince = null;
  // This countdown is not displayed. Its ticking value must not replace the
  // input or focus when the user returns from LINE to read the same PIN.
  const displayState = {...state, lineCheck: state.lineCheck
    ? {...state.lineCheck, retryAfterMs: 0} : state.lineCheck};
  const fingerprint = JSON.stringify(displayState) + lineCode + liveLineCode();
  const linkFingerprint = fingerprint + Boolean(lineConfirmLink);
  if (!force && linkFingerprint === rendered) return;
  // A rate limit becoming available must not erase a live PIN being entered.
  // Preserve only this same store/challenge, never a resent or expired PIN.
  const oldPin = document.querySelector("#pin");
  const pinEntry = !lineConfirmLinkFresh && oldPin && state.lineCheck?.pinAvailable && previous?.lineCheck?.pinAvailable &&
    previous.store?.id === state.store?.id && previous.lineCheck.expiresAt === state.lineCheck.expiresAt
    ? {value:oldPin.value,focused:document.activeElement===oldPin,start:oldPin.selectionStart,end:oldPin.selectionEnd} : null;
  // Status polling must not collapse account controls while they are in use.
  // Keep this only within the same step/store; consent is never persisted.
  const accountActionsOpen = previous?.state === state.state && previous?.store?.id === state.store?.id &&
    document.querySelector("#setup-account-actions")?.open;
  rendered = linkFingerprint;
  await render();
  const accountActions = document.querySelector("#setup-account-actions");
  if (accountActionsOpen && accountActions) accountActions.open = true;
  lineConfirmLinkFresh = false;
  const currentPin = document.querySelector("#pin");
  if (pinEntry && currentPin) {
    currentPin.value = pinEntry.value;
    if (pinEntry.focused) {currentPin.focus();currentPin.setSelectionRange(pinEntry.start,pinEntry.end);}
  }
}
async function mutate(path, body = {}) {
  await api(path, body);
  await refresh(true);
}
async function code() {
  // A new registration starts a new check; an earlier lost number-only send
  // must not be bound to the card that this registration sends.
  forgetPinOnlyAttempt();
  const result = await api("line/code", {});
  lineCode = result.code;
  lineCodeExpiresAt = result.expiresAt;
  await refresh(true);
}

      if (sameBrowser && state.lineButtonConfirm && !fromLink && !state.lineOneButton) {
        const switchButton=button("ボタン確認へ切り替えて送る",async()=>{
          try {await api("line/button",{});}catch(error){await refresh(true);throw error;}
          await refresh(true);
        },"secondary");lockLineSend(switchButton,check,pending);
        const change=node("details",undefined,"setup-recovery");

          node("p","切り替えると前の番号は無効になり、新しい送信を利用枠と回数に数えます。","small"),switchButton);
        root.append(change);
      }
      }
      if (!linkMissing) {
      const retry = node("details", undefined, "setup-recovery");
      retry.append(
        node("summary", "通知先のLINEを変更する・登録コードを送り直す"),
        node("p", "下のボタンで新しい登録コードを表示し、通知先にしたいLINEから送ってください。その後、この元ブラウザへ戻って通知先を確認します。", "small"),
        lockLineSend(button("コードを再発行", code, "secondary"),check),
      );
      root.append(retry);
      }
    }
  } else if (["line_verified", "ready"].includes(state.state)) {
    heading("準備ができました");
    text("次に：下の要点を読み、チェックを入れて「利用を開始する」を押してください。", "next-action");

      "対象は直近60日以内で、まだ返信していない口コミです。返信案は1日1回、最大5件ずつLINEに届きます。登録直後には届かず、参加店で分け合う送信枠や処理の状況で遅れる日があります。",

    ])

    checkbox.type = "checkbox";
    checkbox.id = "terms";

    termsError.id = "terms-error";
    termsError.setAttribute("role", "alert");

      if (checkbox.checked) {
        termsError.hidden = true;
        checkbox.removeAttribute("aria-invalid");
        checkbox.removeAttribute("aria-describedby");
      }
    });

    for (const [title, path] of [
      ["利用条件", "/self/terms"],
      ["プライバシー", "/self/privacy"],
    ]) {

      link.href = path;

      link.rel = "noopener";
      policies.append(link, document.createTextNode(" "));
    }

      policies,
      label,
      termsError,
      button("利用を開始する", async () => {
        if (!checkbox.checked) {
          termsError.textContent = "利用条件を確認し、チェックを入れてください。";
          termsError.hidden = false;
          checkbox.setAttribute("aria-invalid", "true");
          checkbox.setAttribute("aria-describedby", termsError.id);
          checkbox.focus();
          return;
        }
        await mutate("activate", {
          termsVersion: state.termsVersion,
          confirmed: true,
        });
      }),
    );
  } else if (["active", "paused"].includes(state.state)) {
    heading(state.state === "active" ? "稼働中" : "停止中");
    text(state.store.title);
    text(
      state.lastCheckedAt
        ? "最後に口コミの確認を試みた日時：" + new Date(state.lastCheckedAt).toLocaleString("ja-JP")

    );
    if (state.state === "active")
      text("返信案ができたときだけ、原則1日1回、朝9時以降にまとめてLINEへお知らせします。登録直後には届きません。利用枠や処理の状況で、届かない日もあります。", "small");
    text(
      state.processingCount
        ? "処理待ちの口コミ: " +
            state.processingCount +
            "件。利用枠や接続の確認後に順次処理します。"
        : state.scanIncomplete
          ? "口コミ一覧を引き続き取得しています。"
          : state.pendingCount
            ? "承認待ちの返信案: " +
              state.pendingCount +
              "件。LINEで内容を確認してください。"
            : state.replyResults?.length
              ? "各口コミの処理結果を下に表示しています。"
              : state.lastCheckedAt
                ? "未返信の新しい口コミはありません。"
                : state.state === "active"
                  ? "まだ口コミを確認していません。確認は自動で行います。"
                  : "まだ口コミを確認していません。",
      "small",
    );

      text("一部の処理を保留中です。接続や利用枠を確認してください。");
    const help = node("p", undefined, "small");
    const faqLink = node("a", "困ったときは（返信案が届かない・Googleに出ない）");
    faqLink.href = "/self/help/faq#replies";
    help.append(faqLink);
    root.append(help);
    if (state.state === "active")
      root.append(
        button("利用を停止", () => dangerConfirm("pause"), "secondary"),
      );
    else
      root.append(
        button("利用を再開", () =>
          mutate("activate", {
            confirmed: true,
            termsVersion: state.termsVersion,
          }),
        ),
      );
    root.append(
      button("接続を切断", () => dangerConfirm("disconnect"), "danger"),
    );
  }
  if (state.replyResults?.length) {
    root.append(node("h3", "最近の返信結果"));
    const labels = {
      pending: "承認待ち",
      posting: "投稿結果を確認中",
      post_unknown: "投稿結果を照合中",
      reapproval_required: "未投稿を確認。LINEで再承認してください",
      posted: "投稿を確認しました",
      conflict: "変更・別の返信があるため上書きしませんでした",
      skipped: "スキップしました",
      expired: "有効期限が切れました",
      cancelled: "古い通知を無効にしました",
    };
    const list = node("ul", undefined, "reply-results");
    for (const result of state.replyResults) {
      const item = node("li");
      item.append(
        node("strong", labels[result.state] ?? "処理を確認中"),
        node("p", new Date(result.updatedAt).toLocaleString("ja-JP"), "small"),
      );
      if (result.canView)
        item.append(
          link("口コミと返信案を確認", "/account/replies/" + result.id),
        );
      list.append(item);
    }
    root.append(list);
  }
  if (state.lastError === "DRAFT_STORAGE_HELD")
    text(
      "返信案の保存に失敗したため自動再生成を止めています。再生成を希望する場合は一度停止して再開してください。下書きの利用枠を再び消費します。",
      "small",
    );
  if (state.state !== "anonymous") {
    const ordinarySetup = !state.pilotMode && state.registrationAllowed &&
      ["google_connected", "location_selected", "line_pending", "line_verified", "ready"].includes(state.state);
    const accountActions = ordinarySetup ? node("details", undefined, "setup-recovery") : root;
    if (ordinarySetup) {
      accountActions.id = "setup-account-actions";
      accountActions.append(node("summary", "接続の切断・ログアウト"));
      root.append(accountActions);
    }
    if (!["active", "paused"].includes(state.state))
      accountActions.append(
        button("接続を切断", () => dangerConfirm("disconnect"), "danger"),
      );
    accountActions.append(
      button(
        "ログアウト",
        async () => {
          await api("logout", {});
          lineCode = null;
          location.assign("/start");
        },
        "secondary",
      ),
    );
  }
  if (["location_selected", "line_pending"].includes(state.state) && !state.lineDetected) {
    text("LINEから登録コードをまだ受け取っていません。", "small");
    text("通常は自動で検出します。画面が変わらないときだけ、状態を更新してください。", "small");
  }
  if (state.state !== "anonymous")
    root.append(button("状態を更新", () => refresh(true), "secondary"));
  if (state.notice) showError({ code: state.notice });
}
