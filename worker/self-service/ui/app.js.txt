function node(tag, text, cls) {
  const e = document.createElement(tag);
  if (text !== undefined) e.textContent = text;
  if (cls) e.className = cls;
  return e;
}
function showError(error) {
  errorBox.hidden = false;
  errorBox.textContent =
    (state?.lineButtonConfirm && state?.lineCheck?.method === "button" ? buttonMessages[error.code] : null) ?? messages[error.code] ??
    "処理を完了できませんでした。状態を更新してやり直してください。";
  errorBox.tabIndex = -1;
  errorBox.focus();
}
async function api(path, body) {
  const response = await fetch("/api/self/" + path, {
    method: body === undefined ? "GET" : "POST",
    headers:
      body === undefined
        ? {}
        : {
            "Content-Type": "application/json",
            "X-CSRF-Token": state?.csrf ?? "",
          },
    body: body === undefined ? undefined : JSON.stringify(body),
    credentials: "same-origin",
  });
  const data = await response.json();
  if (!response.ok || !data.ok) throw data;
  return data;
}
function button(text, action, cls = "") {
  const b = node("button", text, cls);
  b.type = "button";
  b.disabled = busy;
  b.onclick = async () => {
    if (busy) return;
    busy = true;
    document.querySelectorAll("button").forEach((x) => (x.disabled = true));
    errorBox.hidden = true;
    try {
      await action();
    } catch (e) {
      showError(e);
    } finally {
      busy = false;
      document
        .querySelectorAll("button")
        .forEach(
          (x) => (x.disabled = Boolean((x.dataset.oauth && !challengeToken) || x.dataset.locked === "true")),
        );
    }
  };
  return b;
}

function link(label, url) {
  const a = node("a", label, "button");
  a.href = url;

  return a;
}
function lineChatUrl() {
  try {
    const url = new URL(state.lineFriendUrl);
    const match = /^\/R\/ti\/p\/([^/]+)\/?$/.exec(url.pathname);
    if (url.origin !== "https://line.me" || url.username || url.password ||
        url.search || url.hash || !match) return null;
    const id = decodeURIComponent(match[1]);
    if (!/^@[A-Za-z0-9._-]{1,64}$/.test(id)) return null;
    return "https://line.me/R/oaMessage/" + encodeURIComponent(id);
  } catch { return null; }
}
function liveLineCode() {
  return typeof lineCode === "string" && /^MEOS-[A-HJ-NP-Z2-9]{12}$/.test(lineCode) &&
    Number.isFinite(lineCodeExpiresAt) && lineCodeExpiresAt > Date.now();
}

async function loadChallenge() {
  if (!state.turnstileSiteKey || challengeToken) return;
  const target = document.querySelector("#challenge");
  if (target.dataset.loaded) return;
  target.dataset.loaded = "true";
  const render = () => {
    challengeWidget = window.turnstile.render(target, {
      sitekey: state.turnstileSiteKey,
      action: "self_start",
      callback: (t) => {
        challengeToken = t;
        document
          .querySelectorAll("[data-oauth]")
          .forEach((b) => (b.disabled = busy));
      },
      "expired-callback": clearChallenge,
      "error-callback": clearChallenge,
    });
  };
  if (window.turnstile) return render();
  const script = document.createElement("script");
  script.src =
    "https://challenges.cloudflare.com/turnstile/v0/api.js?render=explicit";
  script.onload = render;
  script.onerror = () => showError({ code: "CHALLENGE_FAILED" });
  document.head.append(script);
}
function clearChallenge() {
  challengeToken = "";
  document.querySelectorAll("[data-oauth]").forEach((b) => (b.disabled = true));
}
function oauthButton(label, intent, cls = "") {
  const b = button(label, () => google(intent), cls);
  b.dataset.oauth = "true";
  b.disabled = !challengeToken;
  return b;
}
async function google(intent) {
  if (!challengeToken) throw { code: "CHALLENGE_FAILED" };
  const token = challengeToken;
  clearChallenge();
  try {
    const r = await api("google/start", { intent, challenge: token });
    location.assign(r.authorizationUrl);
  } catch (error) {
    clearChallenge();
    if (window.turnstile && challengeWidget !== null)
      window.turnstile.reset(challengeWidget);
    throw error;
  }
}
async function refresh(force = false) {
  state = await api(returningFromLine ? "line/return-status" : "status");
  if (state.lineDetected || !["location_selected", "line_pending"].includes(state.state)) {
    lineCode = null;
    lineCodeExpiresAt = 0;
  }
  if (state.lineAutoPin && state.lineDetected && state.lineCheck?.state === "idle") {
    lineIdleSince ??= Date.now();
    state.lineCheck = {...state.lineCheck,
      state: Date.now()-lineIdleSince < 90000 ? "awaiting" : "held",
      code: Date.now()-lineIdleSince < 90000 ? null : "LINE_RETRY_REQUIRED"};
  } else lineIdleSince = null;
  // This countdown is not displayed. Its ticking value must not replace the
  // input or focus when the user returns from LINE to read the same PIN.
  const displayState = {...state, lineCheck: state.lineCheck
    ? {...state.lineCheck, retryAfterMs: 0} : state.lineCheck};
  const fingerprint = JSON.stringify(displayState) + lineCode + liveLineCode();
  if (!force && fingerprint === rendered) return;
  rendered = fingerprint;
  await render();
}
async function mutate(path, body = {}) {
  await api(path, body);
  await refresh(true);
}
async function code() {
  const result = await api("line/code", {});
  lineCode = result.code;
  lineCodeExpiresAt = result.expiresAt;
  await refresh(true);
}
function progress() {
  const i =
    {
      anonymous: 0,
      needs_google_reconnect: 0,
      google_connected: 1,
      location_selected: 2,
      line_pending: 2,
      line_verified: 3,
      ready: 3,
      active: 3,
      paused: 3,
    }[state.state] ?? 0;
  document
    .querySelectorAll("#steps li")
    .forEach((e, n) => {
      e.hidden = state.pilotMode && n > 1;
      if (n === (state.pilotMode ? Math.min(i, 1) : i))
        e.setAttribute("aria-current", "step");
      else e.removeAttribute("aria-current");
    });
}
async function locations(cursor = null) {
  const result = await api(
    "locations" + (cursor ? "?cursor=" + encodeURIComponent(cursor) : ""),
  );
  const list = document.querySelector("#locations");
  for (const loc of result.locations) {
    const box = node("div", undefined, "location");
    box.append(
      node("h3", loc.title),
      button("この店舗を使う", async () => {
        await api("location", {
          accountId: loc.accountId,
          locationId: loc.locationId,
        });
        if (state.pilotMode) await refresh(true);
        else await code();
      }),
    );
    list.append(box);
  }
  nextCursor = result.nextCursor;
  if (result.partial)
    list.append(
      node(
        "p",
        "一部の店舗を取得できませんでした。Googleの管理権限を確認してください。",
        "small",
      ),
    );
  if (!result.locations.length && !cursor)
    list.append(
      node(
        "p",
        "管理できる店舗が見つかりませんでした。お店を管理しているGoogleアカウントかご確認ください。",
      ),
    );
  document.querySelector("#more")?.remove();
  if (nextCursor) {
    const b = button(
      "ほかの店舗を表示",
      () => locations(nextCursor),
      "secondary",
    );
    b.id = "more";
    list.append(b);
  }
}

      node("h3", "口コミ"),

        help.append(node("p", "LINEが未インストールの場合は案内に沿ってインストールし、同じブラウザの登録タブへ戻ります。コードの期限が切れたら再発行してください。", "small"));
        help.append(node("p", "PCからの登録や、コードが入力されない場合は、下のコードをコピーしてLINEのMEO Harnessトークへ貼り付け、送信します。", "small"));
        const c = node("output", lineCode, "code");
        c.dataset.testid = "line-code";
        help.append(
          c,
          button("登録コードをコピー", async () => {
            if (!liveLineCode()) return refresh(true);
            await navigator.clipboard.writeText(lineCode);
          }, "secondary"),
        );
        if (state.lineFriendUrl) help.append(link("LINEで友だち追加", state.lineFriendUrl));
        const qr = node("img", undefined, "qr");
        qr.src = "/self/assets/line-qr.svg";
        qr.alt = "LINE友だち追加のQRコード";
        help.append(node("p", "別の端末でLINEを開く方：友だち追加QR", "small"), qr,
          button("コードを再発行", code, "secondary"));
        root.append(help);
      } else {
        text("中断・画面更新でコードが消えた場合は、下のボタンで新しいコードを表示します。前のコードは使えなくなります。", "small");
        root.append(button("LINEの登録コードを表示", code));
      }
      root.append(
        button("LINEの検出状態を更新", () => refresh(true), "secondary"),
      );
      text("通常は自動で検出します。画面が変わらないときだけ、検出状態を更新してください。", "small");
    } else {
      const check = state.lineCheck ?? { state: "idle", pinAvailable: false };
      const pending = ["awaiting", "preparing", "sending"].includes(check.state);
      const sameBrowser = check.state !== "browser_mismatch";
      if (sameBrowser && state.lineButtonConfirm && check.method === "button") renderButtonCheck(check);
      else {
      text(state.lineButtonConfirm && !sameBrowser
        ? "LINEを検出しました。元の登録画面で通知先を確認してください。"
        : "LINEを検出しました。6桁の番号で通知先を確認します。", "success");
      if (!sameBrowser) {
        text("設定を始めたブラウザへ戻ってください。そちらを使えない場合は、下の補助欄でコードを再発行し、LINEへ送り直してください。", "next-action");
      } else if (pending) {
        text("確認番号の送信状況を確認しています。届いた番号は、状況の更新後にこの画面へ入力できます。", "next-action");
      } else if (check.pinAvailable) {
        text("次に：LINEに届いている6桁を入力し、「確認する」を押してください。", "next-action");
      } else {
        text(check.state === "sent"
          ? "確認番号の期限が切れたか、照合の上限に達しました。新しい番号を送って確認してください。"
          : "次に：「確認番号をLINEに送る」を押してください。LINEで届いた6桁を読み、このブラウザへ戻って入力します。", "next-action");
      }
      if (check.state === "sent" && check.pinAvailable)
        text("LINEに確認番号を送りました。届いている6桁を入力してください。", "success");
      if (sameBrowser && check.pinAvailable && !state.lineAutoPin && lineChatUrl()) {
        const open = link("LINEで確認番号を見る", lineChatUrl());
        open.classList.add("secondary");
        open.referrerPolicy = "no-referrer";
        root.append(open, lineReturnHelp("番号を読んだら"));
      }
      if (check.code && sameBrowser)
        text(state.lineAutoPin && check.code === "LINE_RESULT_UNKNOWN" && !check.pinAvailable
          ? "LINEへの送信結果を確認できません。現在使える確認番号がありません。新しい番号を送って確認してください。"
          : messages[check.code] ?? "送信を完了できませんでした。状態を確認して再送してください。", "small");
      if (sameBrowser) {
        const send = button("確認番号をLINEに送る", async () => {
          try { await api("line/test", {}); }
          catch (error) { await refresh(true); throw error; }
          await refresh(true);
        }, check.pinAvailable || pending ? "secondary" : "");
        send.dataset.locked = String(pending);
        send.disabled = busy || pending;
        if (state.lineAutoPin && (check.pinAvailable || pending)) {
          const resend = node("details", undefined, "setup-recovery");
          resend.append(node("summary", "確認番号が届かない・期限が切れたとき"),
            node("p", "すでに届いている番号は、送り直さずに入力できます。再送すると前の番号は使えなくなります。", "small"), send);
          root.append(resend);
        } else root.append(send);
      }
      if (sameBrowser && !pending && (check.pinAvailable || !state.lineAutoPin)) {
        const label = node("label", "確認番号");
        label.htmlFor = "pin";
        const pin = node("input");
        pin.id = "pin";
        pin.type = "text";
        pin.inputMode = "numeric";
        pin.autocomplete = "one-time-code";
        pin.maxLength = 6;
        root.append(label, pin, button("確認する", () => mutate("line/verify", { pin: pin.value })));
      }
      text("番号は5分間有効です。送信は自動送信を含め1分に1回、1日3回まで。送信受付と実際の到着は別です。", "small");
      if (sameBrowser && state.lineButtonConfirm) {
        const switchButton=button("ボタン確認へ切り替えて送る",async()=>{
          try {await api("line/button",{});}catch(error){await refresh(true);throw error;}
          await refresh(true);
        },"secondary");switchButton.dataset.locked=String(pending);switchButton.disabled=busy||pending;
        const change=node("details",undefined,"setup-recovery");
        change.append(node("summary","番号を入力せずボタンで確認する"),
          node("p","切り替えると前の番号は無効になり、新しい送信を利用枠と回数に数えます。","small"),switchButton);
        root.append(change);
      }
      }
      const retry = node("details", undefined, "setup-recovery");
      retry.append(
        node("summary", "通知先のLINEを変更する・登録コードを送り直す"),
        node("p", "下のボタンで新しい登録コードを表示し、通知先にしたいLINEから送ってください。その後、この元ブラウザへ戻って通知先を確認します。", "small"),
        button("コードを再発行", code, "secondary"),
      );
      root.append(retry);
    }
  } else if (["line_verified", "ready"].includes(state.state)) {
    heading("準備ができました");
    text("次に：利用条件とプライバシーを確認し、チェックを入れて「利用を開始する」を押してください。", "next-action");
    text(state.store.title + " ／ LINE確認済み");
    text(
      "直近60日の未返信口コミから、最初は5件まで返信案を作ります。通知は原則1日1回。ここで利用開始しても、返信はまだ公開されません。",
    );
    for (const [name, value] of [
      ["月間AI下書き数", state.limits.drafts],
      ["共有LINEの月間送信上限", state.limits.pushes],
    ]) {
      const p = node("div", undefined, "capacity");
      p.append(node("span", name), node("strong", value + "件"));
      root.append(p);
    }
    const label = node("label");
    const checkbox = node("input");
    checkbox.type = "checkbox";
    checkbox.id = "terms";
    label.append(checkbox, document.createTextNode("利用条件を確認しました"));
    const policies = node("p", undefined, "small");
    for (const [title, path] of [
      ["利用条件", "/self/terms"],
      ["プライバシー", "/self/privacy"],
    ]) {
      const link = node("a", title + "（別タブ）");
      link.href = path;
      link.target = "_blank";
      link.rel = "noopener";
      policies.append(link, document.createTextNode(" "));
    }
    root.append(
      policies,
      label,
      button("利用を開始する", async () => {
        if (!checkbox.checked) {
          checkbox.focus();
          return;
        }
        await mutate("activate", {
          termsVersion: state.termsVersion,
          confirmed: true,
        });
      }),
    );
  } else if (["active", "paused"].includes(state.state)) {
    heading(state.state === "active" ? "稼働中" : "停止中");
    text(state.store.title);
    text(
      state.lastCheckedAt
        ? "最終確認: " + new Date(state.lastCheckedAt).toLocaleString("ja-JP")
        : "接続は完了しました。次回の定期確認を待っています。",
    );
    text(
      state.processingCount
        ? "処理待ちの口コミ: " +
            state.processingCount +
            "件。利用枠や接続の確認後に順次処理します。"
        : state.scanIncomplete
          ? "口コミ一覧を引き続き取得しています。"
          : state.pendingCount
            ? "承認待ちの返信案: " +
              state.pendingCount +
              "件。LINEで内容を確認してください。"
            : state.replyResults?.length
              ? "各口コミの処理結果を下に表示しています。"
              : state.lastCheckedAt
                ? "未返信の新しい口コミはありません。接続成功と返信投稿の完了は別です。"
                : "まだ口コミを取得していません。架空の口コミは投稿しません。",
      "small",
    );
    if (state.lastError)
      text("一部の処理を保留中です。接続や利用枠を確認してください。");
    if (state.state === "active")
      root.append(
        button("利用を停止", () => dangerConfirm("pause"), "secondary"),
      );
    else
      root.append(
        button("利用を再開", () =>
          mutate("activate", {
            confirmed: true,
            termsVersion: state.termsVersion,
          }),
        ),
      );
    root.append(
      button("接続を切断", () => dangerConfirm("disconnect"), "danger"),
    );
  }
  if (state.replyResults?.length) {
    root.append(node("h3", "最近の返信結果"));
    const labels = {
      pending: "承認待ち",
      posting: "投稿結果を確認中",
      post_unknown: "投稿結果を照合中",
      reapproval_required: "未投稿を確認。LINEで再承認してください",
      posted: "投稿を確認しました",
      conflict: "変更・別の返信があるため上書きしませんでした",
      skipped: "スキップしました",
      expired: "有効期限が切れました",
      cancelled: "古い通知を無効にしました",
    };
    const list = node("ul", undefined, "reply-results");
    for (const result of state.replyResults) {
      const item = node("li");
      item.append(
        node("strong", labels[result.state] ?? "処理を確認中"),
        node("p", new Date(result.updatedAt).toLocaleString("ja-JP"), "small"),
      );
      if (result.canView)
        item.append(
          link("口コミと返信案を確認", "/account/replies/" + result.id),
        );
      list.append(item);
    }
    root.append(list);
  }
  if (state.lastError === "DRAFT_STORAGE_HELD")
    text(
      "返信案の保存に失敗したため自動再生成を止めています。再生成を希望する場合は一度停止して再開してください。下書きの利用枠を再び消費します。",
      "small",
    );
  if (state.state !== "anonymous") {
    if (!["active", "paused"].includes(state.state))
      root.append(
        button("接続を切断", () => dangerConfirm("disconnect"), "danger"),
      );
    root.append(
      button(
        "ログアウト",
        async () => {
          await api("logout", {});
          lineCode = null;
          location.assign("/start");
        },
        "secondary",
      ),
    );
  }
  root.append(button("状態を更新", () => refresh(true), "secondary"));
  if (state.notice) showError({ code: state.notice });
}

  if (event.persisted) resumeLine();
